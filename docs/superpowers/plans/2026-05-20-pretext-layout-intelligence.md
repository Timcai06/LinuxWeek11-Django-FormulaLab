# Pretext Layout Intelligence Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add Pretext-powered layout intelligence to Formula Lab so long LaTeX text, AI suggestions, and paper export previews can be measured without repeated DOM layout reads.

**Architecture:** Keep KaTeX responsible for formula rendering and PaddleOCR responsible for recognition. Introduce Node as a first-class frontend build layer, but keep Django/Python as the web runtime. Use `@chenglou/pretext` only in the browser UI layer for text height prediction, shrinkwrap summaries, review list stability, and export preview risk checks; bundle it with esbuild into a static browser asset consumed by Django templates.

**Tech Stack:** Django templates, vanilla JavaScript, Node.js, npm, `@chenglou/pretext`, esbuild, KaTeX, Playwright.

---

## Scope

This plan implements the first useful slice of Pretext integration:

```text
P1: Review List layout stability + LaTeX summary shrinkwrap
P2: Result page paper export preview risk check
```

This plan does not implement Project/Batch/FormulaItem database refactors, local LLM, Symbol Intelligence, Crop Studio, or full TeX layout. Those belong to later product-core plans.

- [ ] **Step 4: Create the Pretext browser API source**

Create `frontend/formulas/layout_intelligence.js`:

```javascript
import {
  prepare,
  layout,
  prepareWithSegments,
  measureLineStats,
  walkLineRanges,
} from "@chenglou/pretext";

const preparedCache = new Map();

function cacheKey(text, font, options) {
  return JSON.stringify([text, font, options || {}]);
}

function getPrepared(text, font, options = {}) {
  const key = cacheKey(text, font, options);
  if (!preparedCache.has(key)) {
    preparedCache.set(key, prepare(text, font, options));
  }
  return preparedCache.get(key);
}

function measureTextBlock(text, config = {}) {
  const width = Number(config.width || 320);
  const lineHeight = Number(config.lineHeight || 20);
  const font = config.font || "14px JetBrains Mono, ui-monospace, monospace";
  const options = config.whiteSpace ? { whiteSpace: config.whiteSpace } : {};
  const prepared = getPrepared(String(text || ""), font, options);
  return layout(prepared, width, lineHeight);
}

function measureLatexSummary(text, config = {}) {
  const width = Number(config.width || 420);
  const lineHeight = Number(config.lineHeight || 18);
  const font = config.font || "13px JetBrains Mono, ui-monospace, monospace";
  const prepared = prepareWithSegments(String(text || ""), font, { whiteSpace: "pre-wrap" });
  return measureLineStats(prepared, width, lineHeight);
}

function findTightWidth(text, config = {}) {
  const maxWidth = Number(config.maxWidth || 520);
  const minWidth = Number(config.minWidth || 160);
  const lineHeight = Number(config.lineHeight || 18);
  const font = config.font || "13px JetBrains Mono, ui-monospace, monospace";
  const prepared = prepareWithSegments(String(text || ""), font, { whiteSpace: "pre-wrap" });
  const targetLineCount = measureLineStats(prepared, maxWidth, lineHeight).lineCount;
  let low = minWidth;
  let high = maxWidth;

  while (high - low > 4) {
    const mid = Math.floor((low + high) / 2);
    const lineCount = measureLineStats(prepared, mid, lineHeight).lineCount;
    if (lineCount > targetLineCount) {
      low = mid + 1;
    } else {
      high = mid;
    }
  }

  const stats = measureLineStats(prepared, high, lineHeight);
  return {
    width: high,
    lineCount: stats.lineCount,
    maxLineWidth: stats.maxLineWidth,
    targetLineCount,
  };
}

function collectLineWidths(text, config = {}) {
  const width = Number(config.width || 420);
  const font = config.font || "13px JetBrains Mono, ui-monospace, monospace";
  const prepared = prepareWithSegments(String(text || ""), font, { whiteSpace: "pre-wrap" });
  const widths = [];
  walkLineRanges(prepared, width, (line) => {
    widths.push(Math.round(line.width));
  });
  return widths;
}

window.FormulaLayout = {
  measureTextBlock,
  measureLatexSummary,
  findTightWidth,
  collectLineWidths,
};
```

(function () {
    function applyMeasuredSummaries() {
        if (!window.FormulaLayout) {
            return;
        }

        document.querySelectorAll("[data-layout-summary]").forEach((node) => {
            const width = Math.max(220, Math.floor(node.getBoundingClientRect().width));
            const metrics = window.FormulaLayout.measureTextBlock(node.textContent, {
                width,
                lineHeight: 20,
                font: "13px D-DIN, Arial Narrow, Arial, sans-serif",
            });
            node.style.minHeight = `${metrics.height}px`;
            node.dataset.layoutLines = String(metrics.lineCount);
        });
    }

    window.addEventListener("load", applyMeasuredSummaries);
    window.addEventListener("resize", applyMeasuredSummaries);
})();
```

function updatePaperFit(value) {
    if (!paperFitPanel || !window.FormulaLayout) {
        return;
    }

    const source = value || "";
    const targetWidth = current === "inline" ? 360 : 520;
    const summary = window.FormulaLayout.measureLatexSummary(source, {
        width: targetWidth,
        lineHeight: 18,
        font: "13px JetBrains Mono, ui-monospace, monospace",
    });
    const tight = window.FormulaLayout.findTightWidth(source, {
        minWidth: 180,
        maxWidth: targetWidth,
        lineHeight: 18,
        font: "13px JetBrains Mono, ui-monospace, monospace",
    });

    const risky = summary.lineCount > 4 || summary.maxLineWidth > targetWidth * 0.96;
    paperFitRisk.textContent = risky ? "CHECK WIDTH" : "READY";
    paperFitPanel.dataset.paperFitStatus = risky ? "review" : "ready";
    paperFitWidth.textContent = `${targetWidth}px`;
    paperFitLines.textContent = String(summary.lineCount);
    paperFitTight.textContent = `${tight.width}px`;
    paperFitMessage.textContent = risky
        ? "This source may be too wide for narrow paper contexts. Consider align mode or manual line breaks."
        : "This source is likely safe for the selected paper export width.";
}
```
